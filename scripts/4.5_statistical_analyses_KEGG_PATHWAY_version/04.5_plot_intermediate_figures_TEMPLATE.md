If full adaptation is too time-consuming, create a minimal plotting script that generates only essential figures:

```python
#!/usr/bin/env python3
"""Minimal plotting script for KEGG data"""

import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
from pathlib import Path
import argparse
from prevalence_utils import get_data_type_suffix

parser = argparse.ArgumentParser()
parser.add_argument('--data-type', required=True, choices=['reactions', 'ko', 'pathway'])
args = parser.parse_args()

BASE_DIR = Path("/n/scratch/users/b/byc014/github/bac_genome_constraint")
suffix = get_data_type_suffix(args.data_type)
OUTPUT_DIR = BASE_DIR / "results/4.5_statistical_analyses_KEGG_PATHWAY_version/04.5_intermediate_figures"
OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

# Load data
global_params = pd.read_csv(
    BASE_DIR / f"results/4.5_statistical_analyses_KEGG_PATHWAY_version/03_global_scaling/global_scaling_params{suffix}.tsv",
    sep='\t'
)

# Plot 1: Distribution of alpha values
plt.figure(figsize=(10, 6))
plt.hist(global_params['alpha_global'], bins=50, edgecolor='black')
plt.xlabel('Alpha (Scaling Exponent)')
plt.ylabel('Count')
plt.title(f'Distribution of Global Scaling Exponents - {args.data_type.upper()}')
plt.savefig(OUTPUT_DIR / f"alpha_distribution{suffix}.pdf")
plt.close()

# Plot 2: Alpha vs R²
plt.figure(figsize=(10, 6))
plt.scatter(global_params['alpha_global'], global_params['r_squared'], alpha=0.5)
plt.xlabel('Alpha (Scaling Exponent)')
plt.ylabel('R²')
plt.title(f'Scaling Exponent vs Fit Quality - {args.data_type.upper()}')
plt.savefig(OUTPUT_DIR / f"alpha_vs_r2{suffix}.pdf")
plt.close()

print(f"Generated minimal plots for {args.data_type}")
```
